// ইউজারের লাইভ লোকেশন খুঁজে বের করার ফাংশন map.locate({setView: true, maxZoom: 16}); function onLocationFound(e) { var radius = e.accuracy / 2; // আপনার বর্তমান লোকেশনে একটি নীল ডট বা মার্কার বসাবে L.marker(e.latlng).addTo(map) .bindPopup("আপনি এখন এখানে আছেন (ত্রিজ্যা: " + radius + " মিটার)").openPopup(); // আপনার চারপাশের এরিয়া গোল করে দেখাবে L.circle(e.latlng, radius).addTo(map); } map.on('locationfound', onLocationFound); // লোকেশন খুঁজে না পেলে এরর মেসেজ দেবে function onLocationError(e) { alert("আপনার লোকেশন খুঁজে পাওয়া যায়নি। দয়া করে ফোনের GPS অন করুন।"); } map.on('locationerror', onLocationError);
